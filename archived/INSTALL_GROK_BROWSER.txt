ПРИМЕР 1: Простой вопрос
────────────────────────────────────────────────────────────────────────────────

from grok_browser_delegate import GrokBrowserDelegate
import asyncio

async def main():
    grok = GrokBrowserDelegate()
    await grok.setup()
    await grok.login_if_needed()

    response = await grok.ask_question("Write Hello World in Python")
    print(response)

    await grok.close()

asyncio.run(main())
