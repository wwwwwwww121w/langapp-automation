# Определить кто должен делать задачу:
from task_router import Task, TaskCategory, TaskComplexity

task = Task(
    title="Напиши документацию",
    category=TaskCategory.DOCUMENTATION,
    complexity=TaskComplexity.SIMPLE,
    priority="medium"
)
print(task)  # → "🤖 [GROK] Напиши документацию"

# Делегировать задачу Grok:
from grok_delegate import GrokDelegate

grok = GrokDelegate()
docs = grok.write_documentation(
    topic="Setup Guide",
    requirements="..."
)

print(docs)  # → Полная документация от Grok
